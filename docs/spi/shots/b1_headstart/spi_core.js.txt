/* =====================================================================================================
   SPARE PART PAGES, CORE (owner: Claude B)
   - spiApi(path, body): the ONLY place that calls fetch for the spare part pages. In a file:// view or with ?stub=1 it answers from
     spiStub() (spi_stub.js, sample data in the shape of the API) and every page shows a "Fixture data" banner.
   - SPI.pages: one entry per page {nav, title, sub, icon, load(), view(data), after(data)}. Registered with spiPage(id, def).
   - Pages need no build.py patch: render() is wrapped here and the navigation group "Spare parts" is added at run time.
   - Helpers: spiN, spiIdr, spiPct, spiCard (chart or table toggle), spiTable (sortable), spiBars (stacked bars), spiKpi, tooltips.
   Rule: a number that the API does not give is shown as "Not in baseline data", never invented.
   ===================================================================================================== */

async function spiApi(path,body,opt){
 const post=body!==undefined&&body!==null,key=post?null:path;
 if(key&&SPI.cache[key]&&!(opt&&opt.fresh))return SPI.cache[key];
 let out;
 if(SPI.fixture){if(typeof spiStub!=="function")throw new Error("No sample data is available in this view.");out=await spiStub(path,body)}
 else{let r;try{r=await fetch(path,{method:post?"POST":"GET",headers:post?{"Content-Type":"application/json"}:undefined,body:post?JSON.stringify(body):undefined})}
  catch(e){throw new Error("The server could not be reached. Check the connection and try again.")}
  let j=null;try{j=await r.json()}catch(e){}
  if(!r.ok)throw new Error((j&&j.error)||("The server answered "+r.status+"."));
  if(j===null)throw new Error("The server did not return data.");out=j}
 if(key)SPI.cache[key]=out;return out}
